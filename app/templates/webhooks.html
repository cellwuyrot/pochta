Как проверять подпись

Каждый запрос приходит с заголовком X-Smtp-Signature вида t=<unix-время>,v1=<hmac>. Подпись — это HMAC-SHA256 от строки "{t}.{тело запроса}" с ключом-секретом. Время в подписи позволяет отклонять старые запросы (защита от replay). 

import hmac, hashlib, time

def verify(secret, raw_body, header, tolerance=300):
    parts = dict(p.split("=", 1) for p in header.split(","))
    ts, given = int(parts["t"]), parts["v1"]
    if abs(time.time() - ts) > tolerance:
        return False
    expected = hmac.new(
        secret.encode(),
        f"{ts}.".encode() + raw_body,
        hashlib.sha256,
    ).hexdigest()
    return hmac.compare_digest(expected, given)
