Шаг 1: Добавьте DNS записи

  1. Зайдите в панель управления DNS вашего домена (Cloudflare, REG.RU, Timeweb, Beget и т.д.)
  2. Добавьте SPF запись (TXT) — разрешает серверу отправлять письма от имени домена
  3. Добавьте DMARC запись (TXT) — определяет политику обработки неавторизованных писем
  4. Добавьте MX запись — если хотите принимать входящие письма на сервер
  5. Нажмите кнопку «Проверить DNS» выше — система проверит что записи настроены

Добавьте в .env вашего проекта:

SMTP_API_KEY={{ site.api_key }}

Шаг 4: Отправка писем (curl)

curl -X POST http://{{ server_ip }}:3005/api/emails \
  -H "Content-Type: application/json" \
  -H "Authorization: Bearer {{ site.api_key }}" \
  -d '{
    "from_email": "noreply@{{ site.domain }}",
    "to": "user@example.com",
    "subject": "Код подтверждения",
    "html": "<p>Ваш код: <b>123456</b></p>"
  }'

Шаг 5: Интеграция в Next.js / Node.js

// lib/email.ts
export async function sendVerificationCode(email: string, code: string) {
  const res = await fetch(`${process.env.SMTP_SERVICE_URL}/api/emails`, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'Authorization': `Bearer ${process.env.SMTP_API_KEY}`,
    },
    body: JSON.stringify({
      from_email: 'noreply@{{ site.domain }}',
      to: email,
      subject: 'Код подтверждения',
      html: `<p>Ваш код: <b>${code}</b></p>`,
    }),
  });
  return res.json();
}
